DEFINE CONSTANT error_cond = 101.0

FUNCTION remove_newline(str)
    FOR each character in str
        IF character is '\n' OR character is '\r'
            SET character to '\0'
    END FOR
END FUNCTION

FUNCTION uniletters(str)
    FOR each character in str
        SET character to lowercase of character
    END FOR
END FUNCTION

FUNCTION count_newline(str, file)
    SET counter to 0
    WHILE read line from file into str
        IF last character in str is '\n' OR '\r'
            INCREMENT counter
        END IF
    END WHILE
    RETURN counter
END FUNCTION

FUNCTION decprintf(result, file)
    IF result has decimal part
        PRINT result formatted to 2 decimal places to file
    ELSE
        PRINT result as integer to file
    END IF
END FUNCTION

FUNCTION str_to_num(operation, size)
    INITIALIZE num as array of doubles with size
    SET i to 0
    TOKENIZE operation into token
    
    IF token is NULL
        SET num[0] to error_cond
        RETURN num
    END IF

    WHILE token is not NULL
        IF token contains invalid number format
            SET num[0] to error_cond
            RETURN num
        END IF
        CONVERT token to double and store in num[i]
        
        IF num[i] is out of range OR invalid token
            SET num[0] to error_cond
            BREAK
        END IF

        INCREMENT i
        TOKENIZE to next token
    END WHILE

    IF size is 2 AND i is not 2
        SET num[0] to error_cond
    ELSE
        SET size to i
    END IF

    RETURN num
END FUNCTION
